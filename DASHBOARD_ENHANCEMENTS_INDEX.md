## 🔄 Common Integration Patterns

### Pattern 1: Full Dashboard Replacement
```jsx
import { DashboardTemplateSelector } from '@/components/DashboardTemplates';

function Dashboard() {
  return <DashboardTemplateSelector {...props} />;
}
```

### Pattern 2: Add Components to Existing Dashboard
```jsx
import { MetricCard, ResponsiveGrid } from '@/components/DashboardCards';

function ExistingDashboard() {
  return (
    <ResponsiveGrid>
      {metrics.map(m => <MetricCard {...m} />)}
    </ResponsiveGrid>
  );
}
```

### Pattern 3: Tabbed Dashboard
```jsx
import { TabsLayout } from '@/components/DashboardLayouts';
import { DashboardTemplateSelector } from '@/components/DashboardTemplates';

function App() {
  return (
    <TabsLayout tabs={[
      { label: 'Template A', content: <TemplateA /> },
      { label: 'Template B', content: <TemplateB /> },
    ]} />
  );
}
```
